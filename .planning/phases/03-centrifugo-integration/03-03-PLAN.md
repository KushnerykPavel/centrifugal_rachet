                   return

                   log.Printf("bob-pq: Decrypt: %v", err)
                   return

               echoText := "echo: " + string(plain)
               echoMsg, err := sess.Encrypt([]byte(echoText))
               if err != nil {
                   log.Printf("bob-pq: Encrypt echo: %v", err)
                   return
               }
               raw, err := protocol.MarshalEnvelope(protocol.TypeRatchetMsg, echoMsg)
               if err != nil {
                   log.Printf("bob-pq: MarshalEnvelope echo: %v", err)
                   return
               }
               if err := cl.Publish(context.Background(), sub, raw); err != nil {
                   log.Printf("bob-pq: Publish echo: %v", err)
                   return
               }
               n := atomic.AddInt32(&echoCount, 1)
               if n == 5 {
                   log.Printf("bob-pq: sent 5 echoes, exiting")

           }
       }()

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| ch-pq channel → OnPublication data | Raw bytes from ch-pq; json.Unmarshal is the trust boundary |
| pqxdh.PrekeyBundle pointer fields | OneTimePreKey *[32]byte and OPKID *uint32 may be null in JSON; library handles nil correctly |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-3-03-01 | Availability | cbQueue deadlock (PQ binaries) | mitigate | Every OnPublication handler body in bob-pq and alice-pq wrapped in `go func()` per D-10 |
| T-3-03-02 | Tampering | sess nil before initial_msg (PQ) | mitigate | nil guard before sess.Decrypt in TypeRatchetMsg handler in both PQ binaries |
| T-3-03-03 | Integrity (race) | echoCount/recvCount in PQ binaries | mitigate | sync/atomic.AddInt32 for both counters; returned value used for comparison |
| T-3-03-04 | Tampering | malformed pqxdh.PrekeyBundle JSON | mitigate | json.Unmarshal error checked; goroutine returns; Fatalf on handshake error |
| T-3-03-05 | Availability | unbuffered bundleCh in alice-pq | mitigate | bundleCh is make(chan []byte, 1) with select+default — Pitfall 2 avoided |
| T-3-03-06 | Elevation | ch-pq ↔ ch-classical cross-publish | mitigate | PQ binaries reference only protocol.ChannelPQ; classical binaries reference only protocol.ChannelClassical — channel constants are compile-time strings |
</threat_model>
