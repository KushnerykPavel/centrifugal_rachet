   ```go
   go func() {
       var env protocol.Envelope
       if err := json.Unmarshal(data, &env); err != nil {
           log.Printf("bob-classical: unmarshal envelope: %v", err)
           return
       }
       switch env.Type {
       case protocol.TypeInitialMsg:
           var initMsg classical.InitialMessage
           if err := json.Unmarshal(env.Payload, &initMsg); err != nil {
               log.Printf("bob-classical: unmarshal initial_msg: %v", err)
               return
           }
           s, err := classical.ResponderHandshake(priv, initMsg)
           if err != nil {
               log.Printf("bob-classical: ResponderHandshake: %v", err)
               return
           }
           sess = s
           log.Printf("bob-classical: handshake complete")

       case protocol.TypeRatchetMsg:
           if sess == nil {
               log.Printf("bob-classical: session not yet initialized, dropping ratchet_msg")
               return
           }
           var msg classical.Message
           if err := json.Unmarshal(env.Payload, &msg); err != nil {
               log.Printf("bob-classical: unmarshal ratchet_msg: %v", err)
               return
           }
           plain, err := sess.Decrypt(&msg)
           if err != nil {
               log.Printf("bob-classical: Decrypt: %v", err)
               return
           }

           // Unmarshal the decrypted bytes as RatchetPayload (D-05).
           var rp protocol.RatchetPayload
           if err := json.Unmarshal(plain, &rp); err != nil {
               log.Printf("bob-classical: unmarshal ratchet payload: %v", err)
               return
           }
           log.Printf("bob-classical: recv: seq=%d text=%s", rp.Seq, rp.Text)

           // Build echo payload and encrypt (D-05).
           echoRP := protocol.RatchetPayload{Seq: rp.Seq, Text: "echo: " + rp.Text}
           echoJSON, err := json.Marshal(echoRP)
           if err != nil {
               log.Printf("bob-classical: marshal echo payload: %v", err)
               return
           }
           echoMsg, err := sess.Encrypt(echoJSON)
           if err != nil {
               log.Printf("bob-classical: Encrypt echo: %v", err)
               return
           }
           raw, err := protocol.MarshalEnvelope(protocol.TypeRatchetMsg, echoMsg)
           if err != nil {
               log.Printf("bob-classical: MarshalEnvelope echo: %v", err)
               return
           }
           if err := cl.Publish(context.Background(), sub, raw); err != nil {
               log.Printf("bob-classical: Publish echo: %v", err)
               return
           }
           n := atomic.AddInt32(&echoCount, 1)
           if n == 5 {
               log.Printf("bob-classical: sent 5 echoes, exiting")
               os.Exit(0)
           }
       }
   }()
   ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Centrifugo channel → OnPublication data | Raw bytes from ch-classical arrive from the network; json.Unmarshal is the trust boundary |
| sess nil-check | sess is set from one goroutine and read from another; nil check prevents nil dereference on out-of-order messages |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-3-02-01 | Availability | cbQueue deadlock | mitigate | Every OnPublication handler body wrapped in `go func()` per D-10; Publish, Decrypt, Encrypt never called on the cbQueue goroutine |
| T-3-02-02 | Tampering | sess nil before initial_msg | mitigate | nil guard before sess.Decrypt in TypeRatchetMsg handler — drops message with log.Printf instead of panic |
| T-3-02-03 | Integrity (race) | echoCount/recvCount concurrent access | mitigate | sync/atomic.AddInt32 for both counters; returned value used for comparison — no read-modify-write race |
| T-3-02-04 | Tampering | malformed JSON envelope | mitigate | json.Unmarshal error checked; goroutine returns on error — no panic path |
| T-3-02-05 | Spoofing | bob-classical receiving its own prekey_bundle | accept | Centrifugo standard pub/sub does not echo back to publisher (Assumption A2); nil guard on sess handles edge case |
| T-3-02-06 | Availability | unbuffered bundleCh blocking goroutine | mitigate | bundleCh is make(chan []byte, 1) with select+default non-blocking send — Pitfall 2 explicitly avoided |
| T-3-02-07 | Tampering | malformed RatchetPayload JSON inside decrypted bytes | mitigate | json.Unmarshal(plain, &rp) error checked in both bob and alice handlers; goroutine returns on error |
</threat_model>
